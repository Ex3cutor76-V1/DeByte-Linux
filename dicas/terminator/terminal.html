Configuração do terminator:

O terminator é o terminal principal do DeByte, sendo ele o segundo software a ser aberto no DeByte, ele é sua ferramenta mais poderosa e a que você mais vai usar durante o uso do DeByte, afinal, o terminal será sua principal área de trabalho. 

# Configurações globais do Terminator
[global_config]
  borderless = True
  handle_size = 0
  maximise = True
# Atalhos personalizados
[keybindings]
# Perfis que definem a aparência e comportamento dos terminais
[profiles]
  [[default]]
    background_color = "#000000"
    cursor_color = "#ffffff"
    foreground_color = "#ffffff"
    show_titlebar = False
    scrollbar_position = disabled
# Os layouts que descrevem como os terminais são organizados
[layouts]
  [[default]]
    [[[child1]]]
      parent = window0
      type = Terminal
    [[[window0]]]
      parent = ""
      type = Window
# Áreas destinadas aos plugins do Terminator
[plugins]
