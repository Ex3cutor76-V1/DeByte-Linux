# Executa verificações e inicia o tor
tor() {
    if [ -d "/opt/tor-browser" ]; then
        cd /opt/tor-browser && ./start-tor-browser.desktop --detach
        cd ~
    elif ls ~/Downloads/tor-browser*.tar.xz 2>/dev/null | head -1 | grep -q "."; then
        echo "Extraindo Tor Browser..."
        cd ~/Downloads
        tar -xJf tor-browser*.tar.xz
        cd tor-browser
        ./start-tor-browser.desktop --detach
        cd ~
    else
        echo "Tor Browser nao encontrado."
        echo "Baixe em: https://www.torproject.org/download/"
    fi
}

# Mensagem inicial no live system
debyte_welcome() {
VERMELHO=$'\033[31m'
MAGENTA=$'\033[35m'
RESET=$'\033[0m'

    printf "${VERMELHO}"

    cat <<'LOGO'
                       $$$$$$$$$$$$$$$$$$$$$$$$
                 $$$$$$    x              x    $$$$$$
           $$$$$$         x   &   &   &    x         $$$$$$
       $$$$              x     &  &  &      x              $$$$
   $$$$                  x      & & &       x                  $$$$
       $$$$              x        &         x              $$$$
           $$$$$$         x       &        x         $$$$$$
                 $$$$$$    x      &       x    $$$$$$
                       $$$$$$$$$$$$$$$$$$$$$$$$
LOGO

    printf "${RESET}\n"

    echo "                         BEM-VINDO AO DEBYTE"
    printf "  Se quiser usar Tor, use: ${MAGENTA}tor${RESET}\n"
    echo "  Conecte-se à internet com: nmtui ou nmcli dev wifi"
    printf "  Instale no HD com: ${VERMELHO}sudo debyte-install${RESET}\n"
    echo
}

if [ -e /run/live/medium ] && [ -z "$DEBYTE_WELCOME" ]; then
    export DEBYTE_WELCOME=1
    debyte_welcome
fi
